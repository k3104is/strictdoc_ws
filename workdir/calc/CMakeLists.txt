cmake_minimum_required(VERSION 3.20)
project(calc VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(calc_lib src/calculator.cpp src/cli.cpp)
target_include_directories(calc_lib PUBLIC include)
add_executable(calc src/main.cpp)
target_link_libraries(calc PRIVATE calc_lib)
include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz URL_HASH SHA256=7b42b4d6ed48810c5362c265a17faebe90dc2373c885e5216439d37927f02926 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
  add_executable(calc_tests test/calc_test.cpp)
  target_link_libraries(calc_tests PRIVATE calc_lib GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(calc_tests)
endif()
